    <script>
        const monthsThai = [
            "มกราคม", "กุมภาพันธ์", "มีนาคม", "เมษายน", "พฤษภาคม", "มิถุนายน", 
            "กรกฎาคม", "สิงหาคม", "กันยายน", "ตุลาคม", "พฤศจิกายน", "ธันวาคม"
        ];

        let currentDate = new Date();
        let selectedDayKey = null;
        let currentBase64Image = "";

        let tradeData = JSON.parse(localStorage.getItem('tradeData') || '{}');

        function renderCalendar() {
            const year = currentDate.getFullYear();
            const month = currentDate.getMonth();

            document.getElementById('monthYearDisplay').innerText = `${monthsThai[month]} ${year + 543} (ค.ศ. ${year})`;

            const firstDayIndex = (new Date(year, month, 1).getDay() + 6) % 7; 
            const totalDays = new Date(year, month + 1, 0).getDate();

            const grid = document.getElementById('daysGrid');
            grid.innerHTML = '';

            for (let i = 0; i < firstDayIndex; i++) {
                const emptyCell = document.createElement('div');
                emptyCell.className = 'day-cell empty-cell';
                grid.appendChild(emptyCell);
            }

            for (let day = 1; day <= totalDays; day++) {
                const cellKey = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
                const cell = document.createElement('div');
                cell.className = 'day-cell';

                const data = tradeData[cellKey] || { count: '', type: '', pnl: '', bg: '', text: '#ffffff', image: '' };
                
                const hasTradeData = Boolean(data.type || data.pnl !== '' || data.count || data.image);

                // กำหนดสีพื้นหลัง: ถ้าเลือก SL ให้ช่องเป็นสีแดง (#7f1d1d) เสมอ
                if (data.type === 'SL') {
                    cell.style.backgroundColor = '#7f1d1d'; // สีแดงสำหรับ SL
                } else if (data.bg) {
                    cell.style.backgroundColor = data.bg;
                } else if (hasTradeData) {
                    cell.style.backgroundColor = '#7f1d1d';
                } else {
                    cell.style.backgroundColor = '#122217';
                }

                // Format PnL (+ / -)
                let pnlText = '';
                let pnlClass = '';
                if (data.pnl !== '' && data.pnl !== null && !isNaN(data.pnl)) {
                    let val = Number(data.pnl);

                    // ถ้าเป็น SL บังคับให้เป็นค่าลบและติดเครื่องหมาย - นำหน้าเสมอ
                    if (data.type === 'SL') {
                        val = -Math.abs(val);
                    }

                    if (val > 0) {
                        pnlText = `+${val.toLocaleString()}`;
                        pnlClass = 'pnl-plus';
                    } else if (val < 0) {
                        pnlText = `${val.toLocaleString()}`;
                        pnlClass = 'pnl-minus';
                    } else {
                        pnlText = data.type === 'SL' ? '-0' : '0';
                        pnlClass = data.type === 'SL' ? 'pnl-minus' : '';
                    }
                }

                // Badge TP/SL
                let badgeHtml = '';
                if (data.type === 'TP') badgeHtml = `<span class="badge badge-tp">TP</span>`;
                else if (data.type === 'SL') badgeHtml = `<span class="badge badge-sl">SL</span>`;
                else if (data.type === 'EVEN') badgeHtml = `<span class="badge badge-even">EVEN</span>`;

                // Trade count text
                const countText = data.count ? `${Number(data.count).toLocaleString()} ไม้` : '';

                // Chart Icon indicator
                const imageIconHtml = data.image ? `<span class="chart-icon" title="มีรูปกราฟแนบอยู่">📊</span>` : '';

                cell.innerHTML = `
                    <div class="cell-top">
                        <span class="day-number">${day}</span>
                        ${badgeHtml}
                    </div>
                    <div class="cell-body">
                        ${pnlText ? `<div class="pnl-display ${pnlClass}">${pnlText}</div>` : ''}
                        ${countText ? `<div class="trade-count" style="color:${data.text};">${countText}</div>` : ''}
                    </div>
                    <div class="cell-bottom">
                        ${imageIconHtml}
                    </div>
                `;

                cell.onclick = () => openModal(cellKey, day, month, year, data);
                grid.appendChild(cell);
            }
        }

        // เมื่อเปลี่ยนการเลือกผลการเทรด (Status) ใน Modal
        function handleStatusChange() {
            const type = document.getElementById('tradeTypeInput').value;
            const bgInput = document.getElementById('bgColorInput');
            
            if (type === 'SL') {
                bgInput.value = '#7f1d1d'; // ปรับสีตัวเลือกเป็นสีแดงเข้มทันทีที่เลือก SL
            } else if (type === 'TP') {
                bgInput.value = '#14532d'; // ปรับสีตัวเลือกเป็นสีเขียวเมื่อเลือก TP
            }
        }

        function changeMonth(direction) {
            currentDate.setMonth(currentDate.getMonth() + direction);
            renderCalendar();
        }

        function openModal(key, day, month, year, data) {
            selectedDayKey = key;
            currentBase64Image = data.image || "";

            document.getElementById('modalTitle').innerText = `วันที่ ${day} ${monthsThai[month]} ${year}`;
            document.getElementById('tradeTypeInput').value = data.type || "";
            document.getElementById('pnlInput').value = data.pnl !== undefined ? data.pnl : "";
            document.getElementById('tradeCountInput').value = data.count || "";
            
            document.getElementById('bgColorInput').value = data.bg || (data.type === 'SL' ? '#7f1d1d' : '#7f1d1d');
            document.getElementById('textColorInput').value = data.text || "#ffffff";
            document.getElementById('imageInput').value = "";

            updateImagePreviewDisplay();
            document.getElementById('tradeModal').style.display = 'flex';
        }

        function closeModal() {
            document.getElementById('tradeModal').style.display = 'none';
            selectedDayKey = null;
            currentBase64Image = "";
        }

        function previewSelectedImage(input) {
            if (input.files && input.files[0]) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    currentBase64Image = e.target.result;
                    updateImagePreviewDisplay();
                };
                reader.readAsDataURL(input.files[0]);
            }
        }

        function removeImage() {
            currentBase64Image = "";
            document.getElementById('imageInput').value = "";
            updateImagePreviewDisplay();
        }

        function updateImagePreviewDisplay() {
            const imgEl = document.getElementById('imagePreview');
            const removeBtn = document.getElementById('removeImgBtn');
            if (currentBase64Image) {
                imgEl.src = currentBase64Image;
                imgEl.style.display = 'block';
                removeBtn.style.display = 'inline-block';
            } else {
                imgEl.src = '';
                imgEl.style.display = 'none';
                removeBtn.style.display = 'none';
            }
        }

        function saveTradeData() {
            if (!selectedDayKey) return;

            const type = document.getElementById('tradeTypeInput').value;
            let pnl = document.getElementById('pnlInput').value;
            const count = document.getElementById('tradeCountInput').value;
            let bg = document.getElementById('bgColorInput').value;
            const text = document.getElementById('textColorInput').value;

            // ถ้าเลือก SL บังคับให้เปลี่ยนค่า PnL เป็นลบ (-) อัตโนมัติ และตั้งสีพื้นหลังเป็นสีแดง
            if (type === 'SL') {
                if (pnl !== '' && !isNaN(pnl)) {
                    pnl = -Math.abs(Number(pnl));
                }
                bg = '#7f1d1d'; // ตั้งสีเป็นแดงสำหรับ SL
            }

            tradeData[selectedDayKey] = {
                type,
                pnl,
                count,
                bg,
                text,
                image: currentBase64Image
            };

            try {
                localStorage.setItem('tradeData', JSON.stringify(tradeData));
            } catch (e) {
                alert("ไม่สามารถบันทึกได้ เนื่องจากขนาดรูปภาพใหญ่เกินขีดจำกัดของเบราว์เซอร์");
                return;
            }

            closeModal();
            renderCalendar();
        }

        renderCalendar();
    </script>
